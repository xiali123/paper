# ============================================
# PaperCrawler Static Link Configuration
# ============================================
# This file contains CMake configuration for
# building fully static executables
# ============================================

cmake_minimum_required(VERSION 3.15)
project(PaperCrawler VERSION 1.0.0 LANGUAGES CXX)

# Build options
option(BUILD_STATIC "Build static executable (standalone)" ON)
option(BUILD_SHARED "Build shared libraries" OFF)

# C++ Standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Static build configuration
if(BUILD_STATIC)
    message(STATUS "Building STATIC executable (standalone)")

    # Static linking flags
    if(WIN32)
        # Windows static linking
        set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -static")
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libgcc -static-libstdc++")
    else()
        # Linux/Unix static linking
        set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -static-libgcc -static-libstdc++")
        set(CMAKE_FIND_LIBRARY_SUFFIXES ".a")
    endif()

    # Prefer static libraries
    set(CMAKE_LINK_SEARCH_START_STATIC TRUE)
    set(CMAKE_LINK_SEARCH_END_STATIC TRUE)
endif()

# Output directories
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

# Include directories
include_directories(${CMAKE_SOURCE_DIR}/include)

# ============================================
# Dependencies
# ============================================

# Find required packages
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)

# FetchContent for dependencies
include(FetchContent)

# nlohmann/json (header-only, no linking needed)
FetchContent_Declare(
    json
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG v3.11.3
    GIT_SHALLOW TRUE
)
FetchContent_MakeAvailable(json)

# spdlog (header-only if needed)
FetchContent_Declare(
    spdlog
    GIT_REPOSITORY https://github.com/gabime/spdlog.git
    GIT_TAG v1.12.0
    GIT_SHALLOW TRUE
)
FetchContent_MakeAvailable(spdlog)

# gumbo-parser
FetchContent_Declare(
    gumbo
    GIT_REPOSITORY https://github.com/google/gumbo-parser.git
    GIT_TAG v0.10.1
    GIT_SHALLOW TRUE
)
FetchContent_MakeAvailable(gumbo)

# ============================================
# PaperCrawlerCore Library
# ============================================

if(BUILD_SHARED)
    # Build as shared library
    add_subdirectory(core)
else()
    # Build as static library
    file(GLOB_RECURSE CORE_SOURCES
        "${CMAKE_SOURCE_DIR}/src/core/*.cpp"
        "${CMAKE_SOURCE_DIR}/src/network/*.cpp"
        "${CMAKE_SOURCE_DIR}/src/parser/*.cpp"
        "${CMAKE_SOURCE_DIR}/src/database/*.cpp"
    )

    add_library(PaperCrawlerCore STATIC ${CORE_SOURCES})

    target_include_directories(PaperCrawlerCore PUBLIC
        ${CMAKE_SOURCE_DIR}/include
    )

    target_link_libraries(PaperCrawlerCore
        PUBLIC
            nlohmann_json::nlohmann_json
            spdlog::spdlog
            gumbo
            Threads::Threads
            OpenSSL::SSL
            OpenSSL::Crypto
    )

    if(WIN32)
        target_link_libraries(PaperCrawlerCore PUBLIC ws2_32 wsock32)
        target_compile_definitions(PaperCrawlerCore PRIVATE
            WIN32_LEAN_AND_MEAN
            NOMINMAX
            _CRT_SECURE_NO_WARNINGS
        )
    endif()
endif()

# ============================================
# Backend Server Executable
# ============================================

add_subdirectory(backend)

# Ensure backend links statically
if(BUILD_STATIC)
    target_link_libraries(PaperCrawlerServer
        PRIVATE
            PaperCrawlerCore
            nlohmann_json::nlohmann_json
            spdlog::spdlog
            gumbo_static  # Use static version
            Threads::Threads
            OpenSSL::SSL
            OpenSSL::Crypto
    )

    # Additional Windows static libraries
    if(WIN32)
        target_link_libraries(PaperCrawlerServer PRIVATE
            ws2_32
            wsock32
            crypt32  # Windows cryptography
        )
    endif()

    # Optimization flags
    if(MSVC)
        target_compile_options(PaperCrawlerServer PRIVATE /O2 /GL)
        target_link_options(PaperCrawlerServer PRIVATE /LTCG)
    else()
        target_compile_options(PaperCrawlerServer PRIVATE -O3 -flto)
        target_link_options(PaperCrawlerServer PRIVATE -flto)
    endif()
endif()

# ============================================
# Desktop Application
# ============================================

option(BUILD_DESKTOP "Build Qt desktop application" ON)
if(BUILD_DESKTOP)
    add_subdirectory(desktop)
endif()

# ============================================
# Configuration
# ============================================

# Copy config file to build directory
configure_file(
    ${CMAKE_SOURCE_DIR}/config/config.json.example
    ${CMAKE_BINARY_DIR}/config/config.json
    COPYONLY
)

# ============================================
# Installation
# ============================================

install(TARGETS PaperCrawlerServer
    RUNTIME DESTINATION bin
)

install(DIRECTORY config/
    DESTINATION config
    FILES_MATCHING PATTERN "*.json"
)

install(FILES sql/optimize.sql
    DESTINATION sql
)

# ============================================
# Summary
# ============================================

message(STATUS "")
message(STATUS "================== PaperCrawler Configuration ==================")
message(STATUS "Version: ${PROJECT_VERSION}")
message(STATUS "C++ Standard: ${CMAKE_CXX_STANDARD}")
message(STATUS "Build Type: ${CMAKE_BUILD_TYPE}")
message(STATUS "Static Build: ${BUILD_STATIC}")
message(STATUS "Shared Build: ${BUILD_SHARED}")
message(STATUS "Desktop App: ${BUILD_DESKTOP}")
message(STATUS "Compiler: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
message(STATUS "Install Prefix: ${CMAKE_INSTALL_PREFIX}")
message(STATUS "===============================================================")
message(STATUS "")
