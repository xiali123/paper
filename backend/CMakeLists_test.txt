cmake_minimum_required(VERSION 3.15)
project(MySQLConnectionTest)

# 设置C++标准
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 查找MySQL
if(WIN32)
    set(MYSQL_DIR "C:/Program Files/MySQL/MySQL Server 8.0")
    set(MYSQL_INCLUDE_DIR "${MYSQL_DIR}/include")
    set(MYSQL_LIB_DIR "${MYSQL_DIR}/lib")

    # 查找MySQL库文件
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(MYSQL_LIB "${MYSQL_LIB_DIR}/libmysql.lib")
    else()
        set(MYSQL_LIB "${MYSQL_LIB_DIR}/libmysql.lib")
    endif()
else()
    # Linux/Mac
    find_package(MySQL REQUIRED)
    include_directories(${MYSQL_INCLUDE_DIR})
endif()

# 包含目录
include_directories(
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_SOURCE_DIR}/core/external
    ${MYSQL_INCLUDE_DIR}
)

# 添加可执行文件
add_executable(test_mysql_quick test_mysql_quick.cpp)

# 链接MySQL库
target_link_libraries(test_mysql_quick ${MYSQL_LIB})

# Windows特定设置
if(WIN32)
    # 设置运行时库
    set_target_properties(test_mysql_quick PROPERTIES
        LINK_FLAGS "/NODEFAULTLIB:MSVCRT"
    )
endif()
