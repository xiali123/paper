# Standalone End-to-End Test Suite
cmake_minimum_required(VERSION 3.15)
project(EndToEndTestSuite)

# Set C++ standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Include directories
include_directories(
    ${CMAKE_SOURCE_DIR}/../include
    ${CMAKE_SOURCE_DIR}/../include/core
    ${CMAKE_SOURCE_DIR}/../include/business
    ${CMAKE_SOURCE_DIR}/../include/data
    ${CMAKE_SOURCE_DIR}/../include/modules
    ${CMAKE_SOURCE_DIR}/../include/prompts
    ${CMAKE_SOURCE_DIR}/../include/collaboration
)

# Source files
set(TEST_SOURCES
    EndToEndTestSuite.cpp
)

# Create test executable
add_executable(EndToEndTestSuite ${TEST_SOURCES})

# Link libraries (assuming shared libraries are in Release/)
target_link_libraries(EndToEndTestSuite
    ${CMAKE_SOURCE_DIR}/../build/Release/PaperCrawlerCore.lib
    ${CMAKE_SOURCE_DIR}/../build/Release/PaperCrawlerBusiness.lib
    ${CMAKE_SOURCE_DIR}/../build/Release/PaperCrawlerData.lib
)

# Set output directory
set_target_properties(EndToEndTestSuite PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Release
)
